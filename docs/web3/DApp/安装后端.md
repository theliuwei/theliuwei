4. api/portfolio.py
pythonfrom fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from config.database import get_db
from utils.blockchain import get_wallet_balance
from utils.price import get_token_price
import asyncio

router = APIRouter()

@router.get("/portfolio/{wallet_address}")
async def get_portfolio(wallet_address: str, db: Session = Depends(get_db)):
    """获取钱包投资组合"""
    try:
        # 支持的链
        chains = ["ethereum", "bsc", "polygon"]
        portfolio_data = []
        
        for chain in chains:
            # 获取该链上的代币余额
            balances = await get_wallet_balance(wallet_address, chain)
            
            for token_address, balance in balances.items():
                if balance > 0:
                    # 获取代币价格
                    price = await get_token_price(token_address)
                    value = balance * price
                    
                    portfolio_data.append({
                        "chain": chain,
                        "token_address": token_address,
                        "balance": balance,
                        "price_usd": price,
                        "value_usd": value
                    })
        
        return {"wallet_address": wallet_address, "portfolio": portfolio_data}
    
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/price/{token_symbol}")
async def get_price(token_symbol: str):
    """获取代币价格"""
    try:
        price = await get_token_price(token_symbol)
        return {"symbol": token_symbol, "price_usd": price}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

🌐 API数据源集成
utils/price.py（CoinGecko API）
pythonimport httpx
import asyncio

async def get_token_price(token_symbol: str) -> float:
    """从CoinGecko获取代币价格"""
    url = f"https://api.coingecko.com/api/v3/simple/price"
    params = {
        "ids": token_symbol.lower(),
        "vs_currencies": "usd"
    }
    
    async with httpx.AsyncClient() as client:
        response = await client.get(url, params=params)
        data = response.json()
        
        if token_symbol.lower() in data:
            return data[token_symbol.lower()]["usd"]
        return 0.0
